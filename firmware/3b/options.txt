// options.txt for Pi-CMD (CMD HD emulator) on a Raspberry Pi 3B/3B+ with
// hackup's Pi1541io rev 4 interface board and an I2C OLED.
// Board: https://github.com/hackup/Pi1541io
//
// Put this file in the ROOT of the SD card. Uncommented lines are the working
// configuration for that hardware; everything commented out shows the default.
// This lists every option the emulator understands - anything not here is not
// read.

// ---------------------------------------------------------------------------
// Hardware interface
// ---------------------------------------------------------------------------

// Pi1541io is split line ("Option B") hardware.
splitIECLines = 1

// Leave these commented unless your board needs inversion.
// The default suits a 7406; use invertIECOutputs = 0 for a 7407.
//invertIECInputs = 1
//invertIECOutputs = 0

// GPIO used to pull the IEC ATN line low, so the drive can take control of the
// bus and reprogram whichever drive currently owns device 8 or 9 when you press
// SWAP. On a Pi1541io the ATN line reaches the Pi through a BSS138
// bidirectional level shifter, so the pin that reads ATN can drive it too.
// 0 disables it (the HD then still changes its own device number, it just
// cannot move another drive out of the way).
CMDHDAtnOutGPIO = 24

// ---------------------------------------------------------------------------
// CMD HD
// ---------------------------------------------------------------------------

// The boot ROM is found automatically if it is named cmdhd-bootrom.bin in the
// root of the card. Override the name here if you want to.
//CMDHDRomName = cmdhd-bootrom.bin

// The drive's device number lives inside the image (set it with CMD's
// HD-TOOLS in configuration mode). Setting this forces a different number by
// patching the configuration block on the fly. 0 = respect the image.
//
// This only works on an image that HAS a configuration block. A blank one has
// nothing to patch, so the setting is ignored and the drive answers on no
// device number at all until HDOS is installed. Reach a blank drive with
// installation mode instead - SWAP 8 + SWAP 9 while tapping RESET - which
// brings it up on device 30 from the boot ROM.
//CMDHDDeviceID = 0

// Size in MB of the RAM cache in front of the DHD image.
//
// The cache is write behind, not write through: a write is acknowledged to the
// computer as soon as it reaches RAM, and only reaches the card when the
// serial bus goes quiet or the drive is reset. It has to work that way - going
// to the card mid-command freezes the emulated CPU for tens of milliseconds
// and the bus gives up long before that.
//
// So the cache is also a window in which acknowledged writes exist only in the
// Pi's RAM. Pulling the power (or the card) inside that window loses them, and
// the volume is then partly stale rather than obviously broken. Reset now
// flushes, so reset-to-reboot is safe; yanking power is not.
//
// A smaller cache narrows the window and flushes more often, at the cost of
// more trips to the card. Try 4 if you are doing a lot of writing - installing
// GEOS or Wheels onto a partition, say - and want less in flight.
//CMDHDCacheMB = 32

// Front panel buttons. The real CMD HD has four: SWAP 8, SWAP 9,
// WRITE PROTECT and RESET. Button 5 exits emulation (not a CMD HD function).
// Set any to 0 to disable it.
//CMDHDButtonSwap8 = 1
//CMDHDButtonSwap9 = 2
//CMDHDButtonWP = 3
//CMDHDButtonReset = 4
//CMDHDButtonExit = 5

// Mount this image at startup, so the drive comes straight up as a hard drive.
// The file must be in the /cmd-images folder, and this name must match it
// EXACTLY, extension included - .dhd and .img are both valid but they are not
// interchangeable here. A name that does not match is not an error: the drive
// simply never mounts, you are left in the file browser, and because there is
// no browse mode on this branch nothing answers the serial bus at all. If the
// computer sees no drive, check this line against the actual filename first.
AutoMountImage = hd0.dhd
